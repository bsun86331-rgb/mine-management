function findExistingProfile() { const records = getPersonnelRecords(); const currentPersonId = localStorage.getItem( "currentPersonId" ); if ( !currentPersonId || !selectedPosition ) { return null; } /* 必须同时满足： 1. 人员ID相同 2. 当前申请岗位相同 防止一个岗位的人员档案 被另一个岗位申请覆盖。 */ return ( records.find( person => getPersonId(person) === currentPersonId && normalizePosition( person.position ) === selectedPosition ) || null ); }
